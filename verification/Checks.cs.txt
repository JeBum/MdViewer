    const BindingFlags Flags = BindingFlags.Instance | BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;
    static string Root = "", Artifacts = "";
    static readonly List<object> Results = new();
    static int Passed, Failed, Blocked;
    static FileTab Tab = null!;
    static RichTextBox Editor = null!;
    static WebView2 Web = null!;
    static Form Host = null!;
    static object? Call(string name, params object?[] args) => typeof(FileTab).GetMethod(name, Flags)!.Invoke(Tab, args);
    static T Field<T>(string name) => (T)typeof(FileTab).GetField(name, Flags)!.GetValue(Tab)!;
    static void SetField(string name, object value) => typeof(FileTab).GetField(name, Flags)!.SetValue(Tab, value);
    static string Render(string md) => (string)Call("RenderHtml", md)!;
    static string ImageHtml(string md) => (string)Call("RewriteLocalImages", Render(md), Root)!;
    static string Plain(string html) => WebUtility.HtmlDecode(Regex.Replace(html, "<[^>]+>", ""));
    static void Require(bool condition, string detail) { if (!condition) throw new Exception(detail); }
    static void Equal<T>(T expected, T actual) => Require(Equals(expected, actual), $"Expected {JsonSerializer.Serialize(expected)}; actual {JsonSerializer.Serialize(actual)}");
    static void Record(string id, string name, string status, string detail)
    {
        if (status == "PASS") Passed++; else if (status == "FAIL") Failed++; else Blocked++;
        Results.Add(new { id, name, status, detail });
        Console.WriteLine($"{status} {id}: {name} -- {detail}");
        File.WriteAllText(Path.Combine(Artifacts, "results.json"), JsonSerializer.Serialize(Results, new JsonSerializerOptions { WriteIndented = true }));
    }
    static void Test(string id, string name, Action test)
    {
        try { test(); Record(id, name, "PASS", "Assertions satisfied"); }
        catch (Exception e) { Record(id, name, "FAIL", (e.InnerException ?? e).Message); }
    }
    static async Task TestAsync(string id, string name, Func<Task> test)
    {
        try { await test(); Record(id, name, "PASS", "Assertions satisfied"); }
        catch (Exception e) { Record(id, name, "FAIL", (e.InnerException ?? e).Message); }
    }
    static void Text(string value, int? caret = null)
    {
        Editor.Text = value;
        Editor.Select(caret ?? Editor.TextLength, 0);
        Field<System.Windows.Forms.Timer>("_previewTick").Stop();
    }
    static void Key(Keys key) => typeof(Control).GetMethod("OnKeyDown", Flags)!.Invoke(Editor, new object[] { new KeyEventArgs(key) });
    static async Task<JsonElement> JS(string script)
    {
        string result = await Web.ExecuteScriptAsync("(()=>{try{return (" + script + ");}catch(e){return {error:String(e)};}})()");
        return JsonSerializer.Deserialize<JsonElement>(result);
    }
    static async Task Wait(Func<Task<bool>> predicate, int milliseconds = 12000)
    {
        var end = DateTime.UtcNow.AddMilliseconds(milliseconds);
        do { if (await predicate()) return; await Task.Delay(100); } while (DateTime.UtcNow < end);
        throw new TimeoutException("Condition not reached within " + milliseconds + " ms");
    }
    static async Task Page(string md, bool images = false)
    {
        var html = (string)Call("WrapHtml", images ? ImageHtml(md) : Render(md))!;
        File.WriteAllText(Path.Combine(Artifacts, "last-page.html"), html);
        var pageId = Guid.NewGuid().ToString("N");
        html = html.Replace("<head>", "<head><meta name='verification-id' content='" + pageId + "'>");
        Web.NavigateToString(html);
        await Wait(async () => (await JS("document.querySelector('meta[name=verification-id]')?.content")).ToString() == pageId && (await JS("typeof findInPage")).ToString() == "function", 25000);
        await Task.Delay(100);
    }
    [STAThread]
    static int Main(string[] args)
    {
        Root = args[0];
        bool cursorOnly = args.Contains("--cursor");
        Environment.SetEnvironmentVariable("WEBVIEW2_USER_DATA_FOLDER",Path.Combine(Root,"obj","verification","wv2"));
        Artifacts = Path.Combine(Root, "verification", cursorOnly ? "cursor-artifacts" : "artifacts");
        Directory.CreateDirectory(Artifacts);
        Application.SetHighDpiMode(HighDpiMode.SystemAware);
        Application.EnableVisualStyles();
        Application.SetCompatibleTextRenderingDefault(false);
        Brand.Init();
        Host = new Form { Text = "MDviewer verification host", ShowInTaskbar = false, Opacity = 0, Size = new Size(1200, 800) };
        Host.Shown += async (_, _) =>
        {
            try
            {
                if(args.Contains("--native")) { await RunThemeDirtyState(); await RunHomeSample(); return; }
                Tab = new FileTab(null);
                Editor = Field<RichTextBox>("_text");
                if (!cursorOnly) RunLogic();
                Host.Controls.Add(Tab);
                Web = Field<WebView2>("_web");
                await Wait(() => Task.FromResult(Field<bool>("_ready")), 20000);
                if(args.Contains("--editor")){await RunEditorIsolation();return;}
                if (!cursorOnly) { await RunWeb(); await RunApp(); }
                await RunCursor();
                await RunSplit();
                await RunWordView();
                await RunForestTheme();
                await RunColumnMode();
                await RunThemeDirtyState();
                await RunHomeSample();
        await RunHomeAssociationLink();
        await RunHomeThemeLine();
        await RunAssociationDialogColors();
                await RunEditorIsolation();
            }
            catch (Exception e) { Record("ENV", "Integration environment", "BLOCKED", e.ToString()); }
            finally
            {
                if (Tab != null) { Tab.DisposeWatch(); Tab.Dispose(); }
                Console.WriteLine($"TOTAL: {Passed} PASS / {Failed} FAIL / {Blocked} BLOCKED");
                Host.Close();
            }
        };
        Application.Run(Host);
        return Failed > 0 || Blocked > 0 ? 1 : 0;
    }
